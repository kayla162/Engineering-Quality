```python
"""低階活頁簿讀取（.xls 走 xlrd、.xlsx 走 openpyxl）。

ExcelParser 與標單讀取器共用這一層。和 ExcelParser 的差別：這裡**保留欄位位置**
（空格留成 ""），因為標單要知道「第 0 欄是項次、第 6 欄是計價代碼」。
"""

from dataclasses import dataclass
from pathlib import Path

from app.core.errors import UnprocessableDocumentError


@dataclass(frozen=True, slots=True)
class SheetRow:
    sheet: str
    #: 該工作表內的列號（0 起算），換工作表會重新計數
    index: int
    #: 已格式化的字串，**保留欄位位置**，空格是 ""
    cells: list[str]


def format_cell(value: object) -> str:
    """儲存格值轉字串。

    整數型的浮點數要去掉小數尾巴——這不是美觀問題：標準編號、章碼、數量在
    試算表裡常常是數值型別，「3090.0」會讓「CNS 3090」抽不出來。
    """
    if value is None:
        return ""
    if isinstance(value, float) and value.is_integer():
        return str(int(value))
    return str(value).strip()


def read_workbook(path: Path) -> list[SheetRow]:
    if path.suffix.lower() == ".xls":
        return _read_xls(path)
    return _read_xlsx(path)


def _read_xlsx(path: Path) -> list[SheetRow]:
    from openpyxl import load_workbook

    try:
        # data_only=True 取公式的快取值；沒有快取值時是 None，會變成空字串
        book = load_workbook(path, data_only=True, read_only=True)
    except Exception as exc:
        raise UnprocessableDocumentError(f"無法開啟 Excel 檔案：{exc}") from exc

    try:
        return [
            SheetRow(sheet=sheet.title, index=index, cells=[format_cell(c) for c in row])
            for sheet in book.worksheets
            for index, row in enumerate(sheet.iter_rows(values_only=True))
        ]
    finally:
        book.close()


def _read_xls(path: Path) -> list[SheetRow]:
    import xlrd

    try:
        book = xlrd.open_workbook(str(path))
    except Exception as exc:
        raise UnprocessableDocumentError(f"無法開啟 Excel 檔案：{exc}") from exc

    return [
        SheetRow(
            sheet=sheet.name,
            index=index,
            cells=[format_cell(v) for v in sheet.row_values(index)],
        )
        for sheet in book.sheets()
        for index in range(sheet.nrows)
    ]
```

```python
"""Excel 解析（文字化輸出，給法規抽取用）。

為什麼需要這個：工程標單是 Excel，而且真實檔案是 **OLE2 格式的 .xls**——
不是 .xlsx，openpyxl 讀不了。

**一列一個 section，不是一格一個。** DOCX 的表格是一格一段，Excel 照抄的話
真實標單（2669 列 x 7 欄）會產生近兩萬個 section，而且把「壹.一.1.1.1」跟
「施工測量，基地測量及鑑界費」拆散。法規抽取跑在完整的列上準得多。

**邊界：這裡不認得標單的結構。** 項次、計價代碼、哪張工作表是詳細表——那些是
`app/workitems/boq_reader.py` 的職責，它走同一個 `parsers/workbook.py` 讀同一份檔，
但要的是保留欄位位置的結構化資料。
"""

from pathlib import Path

from app.core.enums import ContentType
from app.core.errors import UnprocessableDocumentError
from app.parsers.base import ParsedSection
from app.parsers.workbook import read_workbook

#: 同一列的儲存格用這個串起來。用非中文字元，抽取時才不會把兩格黏成一個法規名稱。
_CELL_SEPARATOR = " | "


def _filled(cells: list[str]) -> list[tuple[int, str]]:
    """回傳 [(欄索引, 文字)]，只留有內容的格子。"""
    return [(index, text) for index, text in enumerate(cells) if text]


def _is_continuation(current: list[tuple[int, str]], previous: list[tuple[int, str]]) -> bool:
    """這一列是不是前一列的換行續寫？

    標單的長說明會跨列續寫，而且是**從詞中間斷開**的（真實資料）：
        列18  項次=''  「新北市建築物施工中管制要點及新北市建」
        列19  項次=''  「築工程施工圍籬綠美化推動實施計畫規定」

    各自成段的話，「新北市建築工程施工圍籬綠美化推動實施計畫」會被切成兩半，
    抽取器就吐出「市建築物施工中管制要點」「勞安標準」這種半截詞。

    判準刻意不綁標單格式（這裡不該認得「項次」欄是什麼）：
    **只有一欄有內容，而且和前一列是同一欄**——那就是換行造成的，不是新的一筆。
    """
    return (
        len(current) == 1
        and len(previous) >= 1
        and current[0][0] in {column for column, _ in previous}
    )


class ExcelParser:
    def parse(self, path: Path) -> list[ParsedSection]:
        sections: list[ParsedSection] = []
        previous_sheet: str | None = None
        previous_cells: list[tuple[int, str]] = []

        for row in read_workbook(path):
            filled = _filled(row.cells)
            if not filled:
                continue

            same_sheet = row.sheet == previous_sheet
            if sections and same_sheet and _is_continuation(filled, previous_cells):
                # 從詞中間斷開的，中間不能插分隔符號，否則名稱還是斷的
                sections[-1].content += filled[0][1]
            else:
                sections.append(
                    ParsedSection(
                        content=_CELL_SEPARATOR.join(text for _, text in filled),
                        order_index=len(sections),
                        content_type=ContentType.TABLE,
                        # 試算表沒有頁碼，跟 DOCX 一樣留 None，不要編一個出來
                        page=None,
                        section_title=row.sheet,
                    )
                )
                previous_cells = filled
            previous_sheet = row.sheet

        if not sections:
            raise UnprocessableDocumentError("試算表裡沒有任何內容")
        return sections
```

```python
from dataclasses import dataclass
from pathlib import Path

from app.core.errors import UnprocessableDocumentError
from app.parsers.workbook import read_workbook

#: 表頭必須同時出現這兩欄才算數（比對前會去掉所有空白）
_REQUIRED_HEADERS = ("項次", "項目及說明")

#: 表頭文字 -> 我們的欄位名。key 是去掉空白後的形式。
_COLUMN_KEYS: dict[str, str] = {
    "項次": "item_no",
    "項目及說明": "name",
    "單位": "unit",
    "數量": "quantity",
    "編碼(備註)": "cost_code",
    "編碼": "cost_code",
}

#: 計價代碼是 10 碼數字，後面可能跟著「,*」之類的記號
_COST_CODE = re.compile(r"(?<!\d)(\d{10})(?!\d)")


class BoqFormatError(UnprocessableDocumentError):
    """標單格式不符預期。"""


@dataclass(slots=True)
class BoqRow:
    item_no: str
    name: str
    unit: str = ""
    quantity: str = ""
    cost_code: str | None = None

    @property
    def is_leaf(self) -> bool:
        """有單位的才是實際工項；「壹」「壹.一」這種是群組列。"""
        return bool(self.unit)


def _normalise(text: str) -> str:
    """去掉所有空白（含全形），表頭才比對得到。"""
    return re.sub(r"\s+", "", text.replace("\u3000", ""))


def _find_header(rows) -> tuple[str, int, dict[str, int]]:
    """回傳 (工作表名, 表頭列號, {欄位名: 欄索引})。"""
    for row in rows:
        normalised = [_normalise(cell) for cell in row.cells]
        if not all(any(key == cell for cell in normalised) for key in _REQUIRED_HEADERS):
            continue
        columns: dict[str, int] = {}
        for index, cell in enumerate(normalised):
            field = _COLUMN_KEYS.get(cell)
            if field is not None and field not in columns:
                columns[field] = index
        return row.sheet, row.index, columns
    raise BoqFormatError(
        "在標單裡找不到表頭（需要同時有「項次」與「項目及說明」兩欄）。"
        "請確認上傳的是詳細價目表，不是總表或單價分析表"
    )


def _cell(cells: list[str], columns: dict[str, int], field: str) -> str:
    index = columns.get(field)
    if index is None or index >= len(cells):
        return ""
    return cells[index]


def read_boq(path: Path) -> list[BoqRow]:
    rows = read_workbook(path)
    sheet, header_index, columns = _find_header(rows)

    items: list[BoqRow] = []
    for row in rows:
        if row.sheet != sheet or row.index <= header_index:
            continue

        item_no = _cell(row.cells, columns, "item_no")
        name = _cell(row.cells, columns, "name")
        if not item_no and not name:
            continue

        if not item_no:
            # 續寫列：只有說明欄有字。從詞中間斷開的，中間不能插分隔符號。
            if items:
                items[-1].name += name
            continue

        raw_code = _cell(row.cells, columns, "cost_code")
        match = _COST_CODE.search(raw_code)
        items.append(
            BoqRow(
                item_no=item_no,
                name=name,
                unit=_cell(row.cells, columns, "unit"),
                quantity=_cell(row.cells, columns, "quantity"),
                cost_code=match.group(1) if match else None,
            )
        )
    return items
```

```python
"""projects / project_documents / work_items 的資料存取。不含業務判斷。"""

import uuid

from sqlalchemy import delete, select
from sqlalchemy.ext.asyncio import AsyncSession

from app.models.project import Project, ProjectDocument, WorkItem


class ProjectRepository:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def create(self, name: str, tenant_id: uuid.UUID | None = None) -> Project:
        project = Project(name=name, tenant_id=tenant_id)
        self.session.add(project)
        await self.session.flush()
        return project

    async def get(self, project_id: uuid.UUID) -> Project | None:
        return await self.session.get(Project, project_id)

    async def list_all(self, limit: int = 100) -> list[Project]:
        result = await self.session.execute(
            select(Project).order_by(Project.created_at.desc()).limit(limit)
        )
        return list(result.scalars().all())

    async def add_document(
        self, project_id: uuid.UUID, document_id: uuid.UUID, role: str
    ) -> ProjectDocument:
        link = ProjectDocument(project_id=project_id, document_id=document_id, role=role)
        self.session.add(link)
        await self.session.flush()
        return link

    async def get_link(
        self, project_id: uuid.UUID, document_id: uuid.UUID
    ) -> ProjectDocument | None:
        result = await self.session.execute(
            select(ProjectDocument).where(
                ProjectDocument.project_id == project_id,
                ProjectDocument.document_id == document_id,
            )
        )
        return result.scalar_one_or_none()

    async def documents_by_role(
        self, project_id: uuid.UUID, role: str
    ) -> list[ProjectDocument]:
        result = await self.session.execute(
            select(ProjectDocument).where(
                ProjectDocument.project_id == project_id, ProjectDocument.role == role
            )
        )
        return list(result.scalars().all())

    async def replace_work_items(
        self, project_id: uuid.UUID, items: list[WorkItem]
    ) -> list[WorkItem]:
        """重建工項清單。重跑標單時要取代而不是累加。"""
        await self.session.execute(delete(WorkItem).where(WorkItem.project_id == project_id))
        self.session.add_all(items)
        await self.session.flush()
        return items

    async def work_items(self, project_id: uuid.UUID) -> list[WorkItem]:
        result = await self.session.execute(
            select(WorkItem)
            .where(WorkItem.project_id == project_id)
            .order_by(WorkItem.order_index)
        )
        return list(result.scalars().all())
```

```python
    async def create_project(self, name: str) -> Project:
        return await self.projects.create(name)

    async def attach_document(
        self, project_id: uuid.UUID, document_id: uuid.UUID, role: DocumentRole
    ) -> ProjectDocument:
        await self._require_project(project_id)
        if await self.projects.get_link(project_id, document_id) is not None:
            raise AppError("這份文件已經掛在這個專案裡了，要換角色請先移除")
        return await self.projects.add_document(project_id, document_id, role)

    async def build_work_items(self, project_id: uuid.UUID) -> list[WorkItem]:
        """從專案的標單重建工項清單。

        只收有單位的明細列——「壹」「壹.一」這種群組列是金額小計，不是工作項目。
        """
        await self._require_project(project_id)

        links = await self.projects.documents_by_role(project_id, DocumentRole.BOQ)
        if not links:
            raise AppError("這個專案還沒有掛標單，沒有工項來源")

        document = await self.documents.get(links[0].document_id)
        if document is None:
            raise AppError("標單文件不存在，可能已被刪除")

        rows = [row for row in read_boq(Path(document.storage_path)) if row.is_leaf]
        items = [
            WorkItem(
                project_id=project_id,
                item_no=row.item_no,
                name=row.name,
                unit=row.unit,
                quantity=row.quantity,
                cost_code=row.cost_code,
                order_index=index,
            )
            for index, row in enumerate(rows)
        ]
        logger.info(
            "從標單建立工項",
            extra=log_context(project_id=str(project_id), work_items=len(items)),
        )
        return await self.projects.replace_work_items(project_id, items)

    async def list_projects(self) -> list[Project]:
        return await self.projects.list_all()

    async def work_items(self, project_id: uuid.UUID) -> list[WorkItem]:
        await self._require_project(project_id)
        return await self.projects.work_items(project_id)

    async def spec_chapter_for(self, cost_code: str | None) -> SpecChapter | None:
        """計價代碼前 5 碼 = 施工綱要規範章碼（實測：有代碼的工項裡 85% 對得到）。

        對不到就回 None——各機關自訂的專業工項不在共通性工項目錄裡，
        硬湊一個章節出來會是「有信心的錯」。
        """
        if not cost_code or len(cost_code) < 5:
            return None
        detail = await self._pcces.get_detail(f"PCCES-{cost_code[:5]}")
        if detail is None:
            return None
        return SpecChapter(code=cost_code[:5], name=detail.name)

    async def _require_project(self, project_id: uuid.UUID) -> Project:
        project = await self.projects.get(project_id)
        if project is None:
            raise ProjectNotFoundError("找不到指定的專案")
        return project
```

```python
"""專案、專案文件角色與工項清單的 API。

工項清單是「工項檢核表」的第一欄，來源是標單的詳細價目表
（docs/superpowers/specs/2026-09-11-工項檢核表-design.md 第 5 節）。
"""

import uuid

from fastapi import APIRouter, status

from app.api.deps import ProjectServiceDep
from app.services.project_service import ProjectService
from app.schemas.project import (
    AttachDocumentRequest,
    ProjectCreate,
    ProjectDocumentOut,
    ProjectListOut,
    ProjectOut,
    WorkItemListOut,
    WorkItemOut,
)

router = APIRouter(prefix="/api/projects", tags=["projects"])


@router.post("", response_model=ProjectOut, status_code=status.HTTP_201_CREATED)
async def create_project(payload: ProjectCreate, service: ProjectServiceDep):
    # 名稱已經由 ProjectCreate 的 validator 去過空白
    return ProjectOut.from_model(await service.create_project(payload.name))


@router.get("", response_model=ProjectListOut)
async def list_projects(service: ProjectServiceDep):
    projects = await service.list_projects()
    return ProjectListOut(
        total=len(projects), items=[ProjectOut.from_model(p) for p in projects]
    )


@router.post(
    "/{project_id}/documents",
    response_model=ProjectDocumentOut,
    status_code=status.HTTP_201_CREATED,
)
async def attach_document(
    project_id: uuid.UUID, payload: AttachDocumentRequest, service: ProjectServiceDep
):
    link = await service.attach_document(project_id, payload.document_id, payload.role)
    return ProjectDocumentOut.from_model(link)


@router.post(
    "/{project_id}/work-items",
    response_model=WorkItemListOut,
    status_code=status.HTTP_201_CREATED,
)
async def build_work_items(project_id: uuid.UUID, service: ProjectServiceDep):
    """從專案的標單重建工項清單。重跑會取代舊的，不會累加。"""
    await service.build_work_items(project_id)
    return await _work_item_list(project_id, service)


@router.get("/{project_id}/work-items", response_model=WorkItemListOut)
async def list_work_items(project_id: uuid.UUID, service: ProjectServiceDep):
    return await _work_item_list(project_id, service)


async def _work_item_list(project_id: uuid.UUID, service: ProjectService) -> WorkItemListOut:
    items = await service.work_items(project_id)
    rows = [
        WorkItemOut.from_model(item, await service.spec_chapter_for(item.cost_code))
        for item in items
    ]
    return WorkItemListOut(
        total=len(rows),
        with_spec_chapter=sum(1 for r in rows if r.spec_chapter is not None),
        items=rows,
    )
```
