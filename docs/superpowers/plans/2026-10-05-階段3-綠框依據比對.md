```python
#: 施工規範頁首格式：<組碼> <章碼>-<頁次> V<版本> <日期>
#: 實測 13.施工規範.pdf：組碼多半等於章碼，但 05500 05503-1 這種不相等的也有
_SPEC_HEADER = "{group} {code}-{page} V{version} 2020/08/19"


def make_spec_pdf_bytes(chapters=None, trailing_pages: int = 0) -> bytes:
    """產生測試用施工規範 PDF。

    `chapters` 是 [(組碼, 章碼, 章名, 版本, 內文)]。`trailing_pages` 是最後面
    沒有頁首的延續頁數量——真實規範書有 24 頁這種（跨頁大表），它們屬於前一章。
    """
    import pymupdf

    chapters = chapters or [("01725", "01725", "施工測量", "3.0", "1.4 相關準則 (1) 測量法 1.5 資料送審")]
    document = pymupdf.open()
    try:
        for group, code, title, version, body in chapters:
            page = document.new_page()
            header = _SPEC_HEADER.format(group=group, code=code, page=1, version=version)
            page.insert_textbox(
                pymupdf.Rect(40, 40, 560, 760),
                f"{header}\n第{code} 章V{version} {title}\n{body}",
                fontsize=10,
                fontname="china-t",
            )
        for _ in range(trailing_pages):
            page = document.new_page()
            page.insert_textbox(
                pymupdf.Rect(40, 40, 560, 760), "附表（續）", fontsize=10, fontname="china-t"
            )
        return document.tobytes()
    finally:
        document.close()
```

- [ ] **Step 5: 建立 `backend/app/specs/chapter_reader.py`**

```python
"""施工規範書的章節切分。

真實規範書的形狀（本案 13.施工規範.pdf，2026-10-05 查證）：
    772 頁、86 章，每頁頁首是 `<組碼> <章碼>-<頁次> V<版本> <日期>`

        01725 01725-1 V3.0 2013/01/10     組碼與章碼相同
        05500 05503-1 V5.0 2020/08/19     組碼與章碼不同，章碼是 05503

    章名在頁首下一行：`第01725 章V3.0 施工測量`

兩個關鍵設計：

1. **不要求每一頁都符合頁首格式。** 實測 748/772 頁（97%）符合，剩下的是封面
   目錄與跨頁大表。沒有頁首的頁歸給前一頁的章節——它們是延續頁。把 regex 調到
   100% 比這條規則脆弱得多。

2. **切不出任何章節要報錯。** 安靜回空清單的結果，是一張全部「請確認依據」的
   檢核表，看起來像「這個案子沒有依據」，實際是「我們沒讀懂規範書」。
   和 `boq_reader` 找不到表頭要報錯是同一條原則。
"""

import re
from dataclasses import dataclass, field
from pathlib import Path

import pymupdf

from app.core.errors import UnprocessableDocumentError

#: 頁首：<組碼> <章碼>-<頁次> V<版本>。章碼是第二組，不是第一組。
_HEADER = re.compile(r"^(\d{5})\s+(\d{5})-(\d+)\s+V([\d.]+)")

#: 章名行：第01725 章V3.0 施工測量
_TITLE = re.compile(r"第\s*\d{5}\s*章\s*V[\d.]+\s*(\S[^\n]*)")


class SpecFormatError(UnprocessableDocumentError):
    """施工規範書格式不符預期。"""


@dataclass(slots=True)
class SpecChapterRow:
    code: str
    title: str
    version: str
    first_page: int
    last_page: int
    pages: list[str] = field(default_factory=list)

    @property
    def text(self) -> str:
        """整章的內文，空白已正規化——「1.4 相關準則」可能跨行。"""
        return " ".join(" ".join(self.pages).split())


def read_spec_chapters(path: Path) -> list[SpecChapterRow]:
    """把施工規範書切成章節。"""
    try:
        document = pymupdf.open(str(path))
    except Exception as exc:
        raise SpecFormatError(f"無法開啟施工規範書：{exc}") from exc

    rows: list[SpecChapterRow] = []
    try:
        for page_index, page in enumerate(document, start=1):
            text = page.get_text()
            header = _HEADER.match(" ".join(text.split()[:4]))
            if header is None:
                # 延續頁：歸給前一章。還沒有任何一章時（封面、目錄）直接略過。
                if rows:
                    rows[-1].pages.append(text)
                    rows[-1].last_page = page_index
                continue

            code, version = header.group(2), header.group(4)
            if rows and rows[-1].code == code:
                rows[-1].pages.append(text)
                rows[-1].last_page = page_index
                continue

            title_match = _TITLE.search(" ".join(text.split()))
            rows.append(
                SpecChapterRow(
                    code=code,
                    title=title_match.group(1).strip() if title_match else "",
                    version=version,
                    first_page=page_index,
                    last_page=page_index,
                    pages=[text],
                )
            )
    finally:
        document.close()

    if not rows:
        raise SpecFormatError(
            "施工規範書裡找不到任何章節頁首（預期格式：章碼 章碼-頁次 V版本）"
        )
    return rows
```

```python
"""章節裡「1.4 相關準則」區塊的定位。

為什麼只取這一段而不是整章：整章內文裡會順帶提到別的規範（施工方法描述、
相關章節交叉引用），抽起來是雜訊。「1.4 相關準則」是規範書自己宣告的依據清單。

實測本案規範書 86 章裡 74 章有這一段；抽出 1220 筆，其中 739 筆是標準
（CNS/ASTM/AWS）、481 筆是法規。標準沒有資料來源可查證，只引述——
這個判斷在 service 層依 `RegulationType` 做，不在這裡。
"""

import re

#: 1.4 相關準則 ... 直到 1.5 或 2. 產品 為止
_BLOCK = re.compile(r"1\.4\s*相關準則(.*?)(?=1\.5\s|2\.\s*產品|$)", re.S)


def basis_block(chapter_text: str) -> str | None:
    """切出章節裡宣告依據的那一段。沒有這一段就回 None。"""
    match = _BLOCK.search(chapter_text)
    if match is None:
        return None
    block = match.group(1).strip()
    return block or None
```

```python
"""spec_chapters / work_item_bases 的資料存取。不含業務判斷。"""

import uuid

from sqlalchemy import delete, select
from sqlalchemy.ext.asyncio import AsyncSession

from app.models.checklist import SpecChapter, WorkItemBasis
from app.models.project import WorkItem


class ChecklistRepository:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def replace_chapters(
        self, document_id: uuid.UUID, chapters: list[SpecChapter]
    ) -> list[SpecChapter]:
        """重建一份規範書的章節索引。重跑取代，不累加。"""
        await self.session.execute(
            delete(SpecChapter).where(SpecChapter.document_id == document_id)
        )
        self.session.add_all(chapters)
        await self.session.flush()
        return chapters

    async def chapters_by_document(self, document_id: uuid.UUID) -> list[SpecChapter]:
        result = await self.session.execute(
            select(SpecChapter).where(SpecChapter.document_id == document_id)
        )
        return list(result.scalars().all())

    async def replace_bases(
        self, work_item_ids: list[uuid.UUID], bases: list[WorkItemBasis]
    ) -> list[WorkItemBasis]:
        """重建一批工項的依據。"""
        if work_item_ids:
            await self.session.execute(
                delete(WorkItemBasis).where(WorkItemBasis.work_item_id.in_(work_item_ids))
            )
        self.session.add_all(bases)
        await self.session.flush()
        return bases

    async def bases_by_project(
        self, project_id: uuid.UUID
    ) -> dict[uuid.UUID, list[WorkItemBasis]]:
        result = await self.session.execute(
            select(WorkItemBasis)
            .join(WorkItem, WorkItem.id == WorkItemBasis.work_item_id)
            .where(WorkItem.project_id == project_id)
        )
        grouped: dict[uuid.UUID, list[WorkItemBasis]] = {}
        for basis in result.scalars().all():
            grouped.setdefault(basis.work_item_id, []).append(basis)
        return grouped
```

```python
"""檢核表：把工項和它的依據組起來，並算出檢核結果。

依據來自本案施工規範書，不是全國共通的 PCCES 目錄——兩者的落差就是
SPEC_NOT_INCLUDED（實測 188 個工項）。

標準（CNS/ASTM/ISO/JIS）只引述不查證：系統沒有這個資料來源，走比對管線會
全部標成「查無此法規」，讓人以為規範引用了不存在的標準。實測把規範書全文
丟進管線，2429 筆引用裡 2263 筆（93%）是 NOT_FOUND，原因就是這個。
"""

import logging
import uuid
from dataclasses import dataclass
from pathlib import Path

from sqlalchemy.ext.asyncio import AsyncSession

from app.ai.factory import get_llm_provider
from app.core.enums import CheckResult, DocumentRole, ParseQuality
from app.core.errors import ProjectNotFoundError
from app.core.logging import log_context
from app.extractors import ExtractionPipeline
from app.models.checklist import SpecChapter, WorkItemBasis
from app.models.project import WorkItem
from app.repositories.checklist import ChecklistRepository
from app.repositories.documents import DocumentRepository
from app.repositories.projects import ProjectRepository
from app.specs import basis_block, read_spec_chapters

logger = logging.getLogger(__name__)


@dataclass(frozen=True)
class ChecklistRow:
    work_item: WorkItem
    result: CheckResult
    chapter: SpecChapter | None
    bases: list[WorkItemBasis]


@dataclass(frozen=True)
class Checklist:
    rows: list[ChecklistRow]
    #: 規範書解析品質不佳時的警告。依據可能殘缺，使用者必須知道。
    spec_warning: str | None = None


class ChecklistService:
    def __init__(self, session: AsyncSession):
        self.session = session
        self.projects = ProjectRepository(session)
        self.documents = DocumentRepository(session)
        self.checklist_repo = ChecklistRepository(session)
        self.extractor = ExtractionPipeline(get_llm_provider())

    async def build(self, project_id: uuid.UUID) -> Checklist:
        """重建整個專案的依據並回傳檢核表。重跑取代。"""
        spec = await self._spec_document(project_id)
        chapter_rows = read_spec_chapters(Path(spec.storage_path)) if spec else []

        chapters: dict[str, SpecChapter] = {}
        if spec:
            models = [
                SpecChapter(
                    document_id=spec.id,
                    code=row.code,
                    title=row.title,
                    version=row.version,
                    first_page=row.first_page,
                    last_page=row.last_page,
                )
                for row in chapter_rows
            ]
            await self.checklist_repo.replace_chapters(spec.id, models)
            chapters = {model.code: model for model in models}

        work_items = await self.projects.work_items(project_id)
        texts = {row.code: row.text for row in chapter_rows}
        bases = await self._build_bases(work_items, chapters, texts)

        await self.session.commit()
        logger.info(
            "建立檢核表",
            extra=log_context(project_id=str(project_id), work_items=len(work_items)),
        )
        return self._assemble(work_items, chapters, bases, spec)

    async def checklist(self, project_id: uuid.UUID) -> Checklist:
        """只讀已經建好的依據，不重建。

        重建要重讀 772 頁的規範書並對 1028 個工項重跑，不是 GET 該做的事；
        而且有副作用的 GET 會讓人很難推理「為什麼數字變了」。
        """
        spec = await self._spec_document(project_id)
        chapters = (
            {c.code: c for c in await self.checklist_repo.chapters_by_document(spec.id)}
            if spec
            else {}
        )
        work_items = await self.projects.work_items(project_id)
        bases = await self.checklist_repo.bases_by_project(project_id)
        return self._assemble(work_items, chapters, bases, spec)

    # --- 內部 ---

    async def _spec_document(self, project_id: uuid.UUID):
        """專案的施工規範書。沒掛就回 None，不報錯——檢核表沒有依據仍然有價值
        （工項都在），報錯會讓使用者什麼都拿不到。這和階段 1「沒掛標單就報錯」
        刻意不對稱：沒有標單時表是空的。"""
        if await self.projects.get(project_id) is None:
            raise ProjectNotFoundError("找不到指定的專案")

        links = await self.projects.documents_by_role(project_id, DocumentRole.SPECIFICATION)
        if not links:
            return None
        return await self.documents.get(links[0].document_id)

    async def _build_bases(
        self,
        work_items: list[WorkItem],
        chapters: dict[str, SpecChapter],
        texts: dict[str, str],
    ) -> dict[uuid.UUID, list[WorkItemBasis]]:
        # 同一章會對到很多工項（實測 01271 一章就 83 個），抽取結果快取
        cache: dict[str, list[tuple[str, str]]] = {}
        bases: list[WorkItemBasis] = []

        for item in work_items:
            code = self._chapter_code(item)
            if code is None or code not in chapters:
                continue
            if code not in cache:
                cache[code] = await self._references_in(texts.get(code, ""))
            for basis_type, reference in cache[code]:
                bases.append(
                    WorkItemBasis(
                        work_item_id=item.id,
                        basis_type=basis_type,
                        reference=reference,
                        detail=f"第{code}章 1.4",
                    )
                )

        await self.checklist_repo.replace_bases([item.id for item in work_items], bases)
        grouped: dict[uuid.UUID, list[WorkItemBasis]] = {}
        for basis in bases:
            grouped.setdefault(basis.work_item_id, []).append(basis)
        return grouped

    async def _references_in(self, chapter_text: str) -> list[tuple[str, str]]:
        block = basis_block(chapter_text)
        if not block:
            return []
        candidates, _ = await self.extractor.extract(block)
        return [(candidate.regulation_type, candidate.raw_text[:512]) for candidate in candidates]

    def _assemble(
        self,
        work_items: list[WorkItem],
        chapters: dict[str, SpecChapter],
        bases: dict[uuid.UUID, list[WorkItemBasis]],
        spec,
    ) -> Checklist:
        rows = [
            ChecklistRow(
                work_item=item,
                result=self._result_for(item, chapters, bases.get(item.id, [])),
                chapter=chapters.get(self._chapter_code(item) or ""),
                bases=bases.get(item.id, []),
            )
            for item in work_items
        ]
        warning = None
        if spec is not None and spec.parse_quality == ParseQuality.DEGRADED:
            warning = f"本案施工規範書的解析品質不佳：{spec.parse_quality_note}。依據可能不完整。"
        return Checklist(rows=rows, spec_warning=warning)

    @staticmethod
    def _chapter_code(item: WorkItem) -> str | None:
        """計價代碼前 5 碼 = 施工綱要規範章碼。"""
        if not item.cost_code or len(item.cost_code) < 5:
            return None
        return item.cost_code[:5]

    def _result_for(
        self, item: WorkItem, chapters: dict[str, SpecChapter], bases: list[WorkItemBasis]
    ) -> CheckResult:
        code = self._chapter_code(item)
        if code is None:
            return CheckResult.NEEDS_CONFIRMATION
        if code not in chapters:
            # 對得到章節但本案規範書沒收錄。完全沒掛規範書時不能這樣說——
            # 那會把「你還沒上傳規範書」指成「規範書漏收這一章」。
            return CheckResult.SPEC_NOT_INCLUDED if chapters else CheckResult.NEEDS_CONFIRMATION
        return CheckResult.OK if bases else CheckResult.NEEDS_CONFIRMATION
```

> **注意：** `_chapter_code` 只看計價代碼，沒有查 PCCES 目錄。查目錄是階段 1 `ProjectService.spec_chapter_for` 的事，這裡要的是「本案規範書有沒有這一章」。兩者不要混。

```python
"""檢核表的 schema。"""

import uuid

from pydantic import BaseModel

from app.core.enums import CheckResult
from app.services.checklist_service import Checklist, ChecklistRow


class BasisOut(BaseModel):
    basis_type: str
    reference: str
    detail: str | None

    #: 標準沒有資料來源可查證，只引述。前端要據此標示，不要讓人以為查證過。
    verified: bool


class ChapterOut(BaseModel):
    code: str
    title: str
    version: str


class ChecklistItemOut(BaseModel):
    work_item_id: uuid.UUID
    item_no: str
    name: str
    result: CheckResult
    chapter: ChapterOut | None
    bases: list[BasisOut]

    @classmethod
    def from_row(cls, row: ChecklistRow) -> "ChecklistItemOut":
        return cls(
            work_item_id=row.work_item.id,
            item_no=row.work_item.item_no,
            name=row.work_item.name,
            result=row.result,
            chapter=(
                ChapterOut(
                    code=row.chapter.code, title=row.chapter.title, version=row.chapter.version
                )
                if row.chapter
                else None
            ),
            bases=[
                BasisOut(
                    basis_type=basis.basis_type,
                    reference=basis.reference,
                    detail=basis.detail,
                    verified=basis.basis_type != "standard",
                )
                for basis in row.bases
            ],
        )


class ChecklistSummaryOut(BaseModel):
    ok: int
    spec_not_included: int
    needs_confirmation: int


class ChecklistOut(BaseModel):
    total: int
    summary: ChecklistSummaryOut
    #: 契約是專案層級的依據，不逐工項列——理由見階段 3 spec 第 3.1 節。
    #: 這句話要跟著檢核表走到畫面與匯出，否則沒有人知道契約欄去哪了。
    contract_note: str
    #: 規範書解析品質不佳時的警告，依據可能殘缺。正常時是 null。
    spec_warning: str | None
    items: list[ChecklistItemOut]

    @classmethod
    def from_checklist(cls, checklist: Checklist) -> "ChecklistOut":
        rows = checklist.rows
        return cls(
            total=len(rows),
            summary=ChecklistSummaryOut(
                ok=sum(1 for r in rows if r.result == CheckResult.OK),
                spec_not_included=sum(
                    1 for r in rows if r.result == CheckResult.SPEC_NOT_INCLUDED
                ),
                needs_confirmation=sum(
                    1 for r in rows if r.result == CheckResult.NEEDS_CONFIRMATION
                ),
            ),
            contract_note=(
                "契約文件優先順序：招標文件內之標價清單，其品項名稱、規格、數量，"
                "優於招標文件內其他文件之內容。契約依據為專案層級，不逐工項列出。"
            ),
            spec_warning=checklist.spec_warning,
            items=[ChecklistItemOut.from_row(row) for row in rows],
        )
```
